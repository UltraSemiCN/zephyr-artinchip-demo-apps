cmake_minimum_required(VERSION 3.20)
project(dashboard_sim C)
set(CMAKE_C_STANDARD 11)
get_filename_component(APP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT LVGL_DIR)
  get_filename_component(LVGL_DIR "${APP_ROOT}/../modules/lib/gui/lvgl" ABSOLUTE)
endif()
# This renderer uses only LVGL's C software backend, without SDL or C++ libraries.
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_DIR}/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_include_directories(lvgl PUBLIC "${LVGL_DIR}" "${LVGL_DIR}/include")
target_compile_definitions(lvgl PUBLIC LV_CONF_PATH="${CMAKE_CURRENT_LIST_DIR}/lv_conf.h")
set(COMMON "${APP_ROOT}/common/dashboard/src")
foreach(profile IN ITEMS dashboard_1024x600_d125_mipi dashboard_720x1280_d125_mipi
                         dashboard_1024x600_d133_lvds dashboard_1024x600_d213_lvds)
  if(profile MATCHES "720x1280")
    set(layout "${APP_ROOT}/applications/${profile}/src/layout.c")
  else()
    set(layout "${COMMON}/layout_1024x600.c")
  endif()
  string(REGEX MATCH "d[0-9]+" chip "${profile}")
  string(TOUPPER "${chip}" chip)
  add_executable(${profile} main.c "${COMMON}/dashboard.c" "${layout}"
                 "${COMMON}/assets/artinchip_logo.c" "${COMMON}/assets/ultrasemi_logo.c")
  target_include_directories(${profile} PRIVATE include "${COMMON}")
  target_compile_definitions(${profile} PRIVATE DASHBOARD_CHIP_NAME="${chip}")
  target_link_libraries(${profile} PRIVATE lvgl m)
endforeach()
